# CV Triage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement automated CV triage — Mac client uploads a zip of PDF CVs, Seraphine backend ranks them with Claude against a job description, sends top 5 + summary to a WhatsApp group/contact, exposes live progress card in chat + dedicated "Trabalhos" tab.

**Architecture:** Frontend Tauri (Mac) zips selected folder client-side and posts multipart to Seraphine backend on VPS. Backend runs an async worker through a 4-phase state machine (extract → rank → send → done) with extraction cascade pdfplumber → Tesseract PT-BR → Claude Vision, ranking via `claude-sonnet-4-5` with prompt-cached rubric, WhatsApp dispatch via new `/api/send-media` route on existing bridge. Progress streamed over existing `/ws/conversation` WebSocket; card persisted as a special chat turn.

**Tech Stack:** FastAPI · SQLite (raw sqlite3) · httpx · pytest + respx · pdfplumber · pytesseract + pdf2image · Anthropic API (Sonnet 4.5) · whatsapp-web.js + multer · Tauri (Rust `zip` crate) · React + WebSocket.

```typescript
import { Router, type Request, type Response } from "express";
import multer from "multer";
import fs from "node:fs/promises";
import { MessageMedia } from "whatsapp-web.js";
import { getClient, getStatus } from "../client.js";

const upload = multer({ dest: "/tmp", limits: { fileSize: 25 * 1024 * 1024 } });
const router = Router();

router.post("/send-media", upload.single("file"), async (req: Request, res: Response) => {
  const { chatId, caption } = req.body as { chatId?: string; caption?: string };
  if (!chatId || !req.file) {
    return res.status(400).json({ error: "missing chatId or file" });
  }
  if (getStatus() !== "ready") {
    if (req.file) await fs.unlink(req.file.path).catch(() => {});
    return res.status(503).json({ error: "bridge not ready" });
  }
  try {
    const media = MessageMedia.fromFilePath(req.file.path);
    media.filename = req.file.originalname;
    const sent = await getClient().sendMessage(
      chatId,
      media,
      caption ? { caption } : undefined,
    );
    return res.json({ ok: true, id: sent.id._serialized });
  } catch (err) {
    return res.status(500).json({ error: (err as Error).message });
  } finally {
    if (req.file) await fs.unlink(req.file.path).catch(() => {});
  }
});

export default router;
```

```python
import sqlite3

_MIGRATIONS: list[tuple[str, str]] = [
    (
        "cv_triage_v1",
        """
        CREATE TABLE IF NOT EXISTS cv_triage_jobs (
          id              INTEGER PRIMARY KEY AUTOINCREMENT,
          conversation_id INTEGER NOT NULL,
          turn_id         INTEGER NOT NULL,
          status          TEXT    NOT NULL,
          rubric          TEXT    NOT NULL,
          target_chat_id  TEXT    NOT NULL,
          target_label    TEXT    NOT NULL,
          total_cvs       INTEGER NOT NULL DEFAULT 0,
          processed_cvs   INTEGER NOT NULL DEFAULT 0,
          ocr_used_count  INTEGER NOT NULL DEFAULT 0,
          vision_used_count INTEGER NOT NULL DEFAULT 0,
          unreadable      INTEGER NOT NULL DEFAULT 0,
          staging_dir     TEXT    NOT NULL,
          top5_json       TEXT,
          whatsapp_status TEXT,
          error           TEXT,
          created_at      TEXT    NOT NULL,
          finished_at     TEXT
        );
        CREATE TABLE IF NOT EXISTS cv_triage_results (
          id            INTEGER PRIMARY KEY AUTOINCREMENT,
          job_id        INTEGER NOT NULL REFERENCES cv_triage_jobs(id) ON DELETE CASCADE,
          filename      TEXT    NOT NULL,
          candidate_name TEXT,
          text_source   TEXT    NOT NULL,
          text_chars    INTEGER NOT NULL DEFAULT 0,
          score         REAL,
          match_pct     REAL,
          justification TEXT,
          strengths     TEXT,
          gaps          TEXT,
          status        TEXT    NOT NULL,
          error         TEXT,
          ranked_at     TEXT,
          UNIQUE(job_id, filename)
        );
        CREATE INDEX IF NOT EXISTS idx_results_job_score
          ON cv_triage_results(job_id, score DESC);
        """,
    ),
]

def apply_migrations(conn: sqlite3.Connection) -> None:
    conn.executescript(
        "CREATE TABLE IF NOT EXISTS schema_migrations (id TEXT PRIMARY KEY, applied_at TEXT NOT NULL);"
    )
    applied = {row[0] for row in conn.execute("SELECT id FROM schema_migrations")}
    for mig_id, sql in _MIGRATIONS:
        if mig_id in applied:
            continue
        with conn:
            conn.executescript(sql)
            conn.execute(
                "INSERT INTO schema_migrations(id, applied_at) VALUES (?, datetime('now'))",
                (mig_id,),
            )
```

```python
import json
import sqlite3
from typing import Optional, Iterable
from .models import (
    Job, Result, JobStatus, WhatsAppStatus, ResultStatus, TextSource,
)

def create_job(conn: sqlite3.Connection, *, conversation_id: int, turn_id: int,
               rubric: str, target_chat_id: str, target_label: str,
               staging_dir: str, created_at: str) -> int:
    with conn:
        cur = conn.execute(
            """INSERT INTO cv_triage_jobs
               (conversation_id, turn_id, status, rubric, target_chat_id,
                target_label, staging_dir, created_at)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?)""",
            (conversation_id, turn_id, JobStatus.QUEUED.value, rubric,
             target_chat_id, target_label, staging_dir, created_at),
        )
    return int(cur.lastrowid)

def get_job(conn: sqlite3.Connection, job_id: int) -> Optional[Job]:
    row = conn.execute("SELECT * FROM cv_triage_jobs WHERE id=?", (job_id,)).fetchone()
    if not row: return None
    return _row_to_job(row)

def list_jobs(conn: sqlite3.Connection, limit: int = 100) -> list[Job]:
    rows = conn.execute(
        "SELECT * FROM cv_triage_jobs ORDER BY id DESC LIMIT ?", (limit,)
    ).fetchall()
    return [_row_to_job(r) for r in rows]

def update_status(conn: sqlite3.Connection, job_id: int, status: JobStatus,
                  *, error: Optional[str] = None,
                  whatsapp_status: Optional[WhatsAppStatus] = None,
                  finished_at: Optional[str] = None,
                  top5_json: Optional[str] = None) -> None:
    fields = ["status=?"]; values: list = [status.value]
    if error is not None: fields.append("error=?"); values.append(error)
    if whatsapp_status is not None:
        fields.append("whatsapp_status=?"); values.append(whatsapp_status.value)
    if finished_at is not None: fields.append("finished_at=?"); values.append(finished_at)
    if top5_json is not None: fields.append("top5_json=?"); values.append(top5_json)
    values.append(job_id)
    with conn:
        conn.execute(f"UPDATE cv_triage_jobs SET {', '.join(fields)} WHERE id=?", values)

def increment_counters(conn: sqlite3.Connection, job_id: int, *,
                       processed: int = 0, ocr: int = 0, vision: int = 0,
                       unreadable: int = 0) -> None:
    with conn:
        conn.execute(
            """UPDATE cv_triage_jobs SET
               processed_cvs = processed_cvs + ?,
               ocr_used_count = ocr_used_count + ?,
               vision_used_count = vision_used_count + ?,
               unreadable = unreadable + ?
               WHERE id=?""",
            (processed, ocr, vision, unreadable, job_id),
        )

def set_total_cvs(conn: sqlite3.Connection, job_id: int, total: int) -> None:
    with conn:
        conn.execute("UPDATE cv_triage_jobs SET total_cvs=? WHERE id=?", (total, job_id))

def mark_active_jobs_failed_on_startup(conn: sqlite3.Connection, reason: str) -> int:
    with conn:
        cur = conn.execute(
            """UPDATE cv_triage_jobs SET status='failed', error=?, finished_at=datetime('now')
               WHERE status IN ('queued','extracting','ranking','sending')""",
            (reason,),
        )
    return cur.rowcount

def upsert_result(conn: sqlite3.Connection, *, job_id: int, filename: str,
                  text_source: str, text_chars: int = 0,
                  candidate_name: Optional[str] = None,
                  score: Optional[float] = None,
                  match_pct: Optional[float] = None,
                  justification: Optional[str] = None,
                  strengths: Optional[list[str]] = None,
                  gaps: Optional[list[str]] = None,
                  status: str = "pending",
                  error: Optional[str] = None,
                  ranked_at: Optional[str] = None) -> None:
    with conn:
        conn.execute(
            """INSERT INTO cv_triage_results
               (job_id, filename, candidate_name, text_source, text_chars, score,
                match_pct, justification, strengths, gaps, status, error, ranked_at)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
               ON CONFLICT(job_id, filename) DO UPDATE SET
                 candidate_name=excluded.candidate_name,
                 text_source=excluded.text_source,
                 text_chars=excluded.text_chars,
                 score=excluded.score,
                 match_pct=excluded.match_pct,
                 justification=excluded.justification,
                 strengths=excluded.strengths,
                 gaps=excluded.gaps,
                 status=excluded.status,
                 error=excluded.error,
                 ranked_at=excluded.ranked_at""",
            (job_id, filename, candidate_name, text_source, text_chars, score,
             match_pct, justification,
             json.dumps(strengths or []), json.dumps(gaps or []),
             status, error, ranked_at),
        )

def get_top_n(conn: sqlite3.Connection, job_id: int, n: int) -> list[Result]:
    rows = conn.execute(
        """SELECT * FROM cv_triage_results
           WHERE job_id=? AND status='ranked'
           ORDER BY score DESC LIMIT ?""",
        (job_id, n),
    ).fetchall()
    return [_row_to_result(r) for r in rows]

def list_results(conn: sqlite3.Connection, job_id: int) -> list[Result]:
    rows = conn.execute(
        "SELECT * FROM cv_triage_results WHERE job_id=? ORDER BY score DESC NULLS LAST",
        (job_id,),
    ).fetchall()
    return [_row_to_result(r) for r in rows]

def _row_to_job(row: sqlite3.Row) -> Job:
    return Job(
        id=row["id"], conversation_id=row["conversation_id"], turn_id=row["turn_id"],
        status=JobStatus(row["status"]), rubric=row["rubric"],
```

```python
import io
import os
import zipfile
from pathlib import Path

class ZipExtractionError(ValueError):
    pass

def extract_pdfs_safely(
    zip_bytes: bytes,
    dest_dir: Path,
    *,
    max_uncompressed_bytes: int = 100 * 1024 * 1024,
) -> list[Path]:
    if not zipfile.is_zipfile(io.BytesIO(zip_bytes)):
        raise ZipExtractionError("not a zip")
    dest_dir.mkdir(parents=True, exist_ok=True)
    dest_root = dest_dir.resolve()
    extracted: list[Path] = []
    with zipfile.ZipFile(io.BytesIO(zip_bytes)) as zf:
        total = sum(i.file_size for i in zf.infolist())
        if total > max_uncompressed_bytes:
            raise ZipExtractionError(
                f"zip too large uncompressed: {total} > {max_uncompressed_bytes}"
            )
        for info in zf.infolist():
            if info.is_dir(): continue
            name = info.filename
            if name.startswith(("__MACOSX/", "._")) or os.path.basename(name) == ".DS_Store":
                continue
            if not name.lower().endswith(".pdf"):
                continue
            if "\x00" in name or name.startswith("/") or ".." in Path(name).parts:
                raise ZipExtractionError(f"unsafe path: {name!r}")
            target = (dest_root / os.path.basename(name)).resolve()
            if os.path.commonpath([str(dest_root), str(target)]) != str(dest_root):
                raise ZipExtractionError(f"unsafe path: {name!r}")
            with zf.open(info) as src, open(target, "wb") as dst:
                dst.write(src.read())
            extracted.append(target)
    return extracted
```

```python
    stripped = (text or "").strip()
```

```python
from dataclasses import dataclass
from pathlib import Path
from typing import Optional, Any
import base64
import logging

from .models import TextSource
from .quality import is_usable

logger = logging.getLogger(__name__)

@dataclass
class ExtractionResult:
    text: str
    source: TextSource

_OCR_TIMEOUT_SEC = 30
_OCR_DPI = 250
_VISION_MAX_PAGES = 2

def extract_with_fallback(pdf_path: Path, *, vision_client: Optional[Any]) -> ExtractionResult:
    # 1) pdfplumber
    try:
        text = _pdfplumber_extract(pdf_path)
    except Exception as e:
        logger.warning("pdfplumber failed for %s: %s", pdf_path.name, e)
        text = ""
    if is_usable(text):
        return ExtractionResult(text, TextSource.PDFPLUMBER)

    # 2) tesseract
    try:
        text = _tesseract_extract(pdf_path)
    except Exception as e:
        logger.warning("tesseract failed for %s: %s", pdf_path.name, e)
        text = ""
    if is_usable(text):
        return ExtractionResult(text, TextSource.TESSERACT)

    # 3) vision
    if vision_client is not None:
        try:
            text = _vision_extract(pdf_path, vision_client)
        except Exception as e:
            logger.warning("vision failed for %s: %s", pdf_path.name, e)
            text = ""
        if is_usable(text):
            return ExtractionResult(text, TextSource.VISION)

    return ExtractionResult("", TextSource.NONE)

def _pdfplumber_extract(pdf_path: Path) -> str:
    import pdfplumber
    chunks: list[str] = []
    with pdfplumber.open(str(pdf_path)) as pdf:
        for page in pdf.pages:
            t = page.extract_text() or ""
            if t: chunks.append(t)
    return "\n".join(chunks)

def _tesseract_extract(pdf_path: Path) -> str:
    import pytesseract
    from pdf2image import convert_from_path
    pages = convert_from_path(str(pdf_path), dpi=_OCR_DPI)
    parts: list[str] = []
    for img in pages:
        parts.append(pytesseract.image_to_string(img, lang="por", timeout=_OCR_TIMEOUT_SEC))
    return "\n".join(parts)

def _vision_extract(pdf_path: Path, vision_client: Any) -> str:
    """Pass first N pages of the PDF as images to Claude for transcription."""
    from pdf2image import convert_from_path
    import io
    pages = convert_from_path(str(pdf_path), dpi=200, first_page=1, last_page=_VISION_MAX_PAGES)
    image_blocks = []
    for img in pages:
        buf = io.BytesIO()
        img.save(buf, format="PNG")
        b64 = base64.standard_b64encode(buf.getvalue()).decode("ascii")
        image_blocks.append({
            "type": "image",
            "source": {"type": "base64", "media_type": "image/png", "data": b64},
        })
    # vision_client is the AnthropicClient from packages/llm-adapters.
    # Call it with a single user turn that contains the image blocks + a transcription prompt.
    # The exact method name (`chat`, `complete`, etc.) must match the adapter — see
    # packages/llm-adapters/src/llm_adapters/anthropic.py for the actual API.
```

```python
import asyncio
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional
import httpx

@dataclass
class DispatchPdf:
    rank: int
    candidate_name: str
    score: float
    path: Path
    filename: str

@dataclass
class DispatchPayload:
    chat_id: str
    summary: str
    pdfs: list[DispatchPdf]

@dataclass
class DispatchResult:
    text_sent: bool = False
    media_sent: int = 0
    error: Optional[str] = None
    failed_ranks: list[int] = field(default_factory=list)

async def dispatch_to_whatsapp(
    payload: DispatchPayload,
    *,
    bridge_url: str,
    sleep_between: float = 1.5,
    retry_backoff_base: float = 2.0,
    timeout: float = 60.0,
) -> DispatchResult:
    result = DispatchResult()
    async with httpx.AsyncClient(timeout=timeout) as http:
        try:
            r = await http.post(
                f"{bridge_url}/api/send",
                json={"chatId": payload.chat_id, "message": payload.summary},
            )
            r.raise_for_status()
            result.text_sent = True
        except Exception as e:
            result.error = f"send text failed: {e}"
            return result

        for pdf in payload.pdfs:
            caption = f"Top {pdf.rank} · {pdf.candidate_name} · score {pdf.score:.0f}"
            sent = False
            last_err: Optional[str] = None
            for attempt in range(3):
                try:
                    with open(pdf.path, "rb") as f:
                        r = await http.post(
                            f"{bridge_url}/api/send-media",
                            files={"file": (pdf.filename, f, "application/pdf")},
                            data={"chatId": payload.chat_id, "caption": caption},
                        )
                    r.raise_for_status()
                    sent = True
                    break
                except Exception as e:
                    last_err = str(e)
                    if attempt < 2 and retry_backoff_base > 0:
                        await asyncio.sleep(retry_backoff_base ** attempt)
            if sent:
                result.media_sent += 1
            else:
                result.failed_ranks.append(pdf.rank)
                if not result.error: result.error = f"media failed: {last_err}"
            await asyncio.sleep(sleep_between)
    return result
```

```python
import asyncio
import json
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Awaitable, Callable, Optional
import logging

from . import storage
from .models import JobStatus, WhatsAppStatus, TextSource, ResultStatus
from .extraction import ExtractionResult
from .ranking import ScoreResult
from .whatsapp import dispatch_to_whatsapp, DispatchPayload, DispatchPdf

logger = logging.getLogger(__name__)

ExtractFn = Callable[..., Awaitable[ExtractionResult]]
RankFn = Callable[..., Awaitable[ScoreResult]]
BroadcastFn = Callable[[dict], Awaitable[None]]

@dataclass
class WorkerDeps:
    db: Any
    bridge_url: str
    llm_client: Any
    vision_client: Optional[Any]
    extract_fn: ExtractFn
    rank_fn: RankFn
    broadcast_fn: BroadcastFn
    rank_concurrency: int = 5
    sleep_between_media: float = 1.5
    sonnet_model: str = "claude-sonnet-4-5"

_TOP_N = 5
_cancel_flags: set[int] = set()

def request_cancel(job_id: int) -> None: _cancel_flags.add(job_id)

def _is_cancelled(job_id: int) -> bool: return job_id in _cancel_flags

def _now() -> str: return datetime.now(timezone.utc).isoformat()

async def run_cv_triage(job_id: int, deps: WorkerDeps) -> None:
    try:
        await _run(job_id, deps)
    except asyncio.CancelledError:
        storage.update_status(deps.db, job_id, JobStatus.CANCELLED, finished_at=_now())
        raise
    except Exception as e:
        logger.exception("cv_triage job %s failed", job_id)
        storage.update_status(deps.db, job_id, JobStatus.FAILED,
                              error=str(e)[:1000], finished_at=_now())
    finally:
        _cancel_flags.discard(job_id)

async def _run(job_id: int, deps: WorkerDeps) -> None:
    job = storage.get_job(deps.db, job_id)
    if job is None: raise RuntimeError(f"job {job_id} not found")

    staging = Path(job.staging_dir)
    pdfs = sorted(p for p in staging.glob("*.pdf"))
    storage.set_total_cvs(deps.db, job_id, len(pdfs))
    storage.update_status(deps.db, job_id, JobStatus.EXTRACTING)
    await deps.broadcast_fn({"type": "cv_triage_progress", "job_id": job_id,
                             "phase": "extracting", "processed": 0,
                             "total": len(pdfs), "ocr_used_count": 0,
                             "vision_used_count": 0})

    # Phase: extract
    extracted: dict[str, tuple[str, TextSource]] = {}
    for i, pdf in enumerate(pdfs, start=1):
        if _is_cancelled(job_id): raise asyncio.CancelledError
        try:
            er = await deps.extract_fn(pdf, vision_client=deps.vision_client)
        except Exception as e:
            er = ExtractionResult("", TextSource.NONE)
            logger.warning("extract failed %s: %s", pdf.name, e)
        if er.source == TextSource.NONE:
            storage.upsert_result(deps.db, job_id=job_id, filename=pdf.name,
                                  text_source=TextSource.NONE.value, text_chars=0,
                                  status=ResultStatus.FAILED.value,
                                  error="extraction failed")
            storage.increment_counters(deps.db, job_id, processed=1, unreadable=1)
        else:
            storage.upsert_result(deps.db, job_id=job_id, filename=pdf.name,
                                  text_source=er.source.value, text_chars=len(er.text),
                                  status=ResultStatus.EXTRACTED.value)
            extracted[pdf.name] = (er.text, er.source)
            inc = {"processed": 1}
            if er.source == TextSource.TESSERACT: inc["ocr"] = 1
            elif er.source == TextSource.VISION: inc["vision"] = 1
            storage.increment_counters(deps.db, job_id, **inc)
        job = storage.get_job(deps.db, job_id)
        await deps.broadcast_fn({"type": "cv_triage_progress", "job_id": job_id,
                                 "phase": "extracting", "processed": job.processed_cvs,
                                 "total": job.total_cvs, "current_filename": pdf.name,
                                 "ocr_used_count": job.ocr_used_count,
                                 "vision_used_count": job.vision_used_count})

    # Phase: rank
    storage.update_status(deps.db, job_id, JobStatus.RANKING)
    sem = asyncio.Semaphore(deps.rank_concurrency)
    ranked_count = 0
    ranked_lock = asyncio.Lock()

    async def _rank_task(filename: str, text: str, source: TextSource) -> None:
        nonlocal ranked_count
        async with sem:
            if _is_cancelled(job_id): return
            try:
                score = await deps.rank_fn(
                    deps.llm_client, rubric=job.rubric, cv_text=text,
                    model=deps.sonnet_model,
                )
                storage.upsert_result(
                    deps.db, job_id=job_id, filename=filename,
                    candidate_name=score.candidate_name,
                    text_source=source.value,
                    text_chars=len(text),
                    score=score.score, match_pct=score.match_pct,
                    justification=score.justification,
                    strengths=score.strengths, gaps=score.gaps,
                    status=ResultStatus.RANKED.value, ranked_at=_now(),
                )
            except Exception as e:
                storage.upsert_result(
                    deps.db, job_id=job_id, filename=filename,
                    text_source=source.value, text_chars=len(text),
                    status=ResultStatus.FAILED.value, error=str(e)[:500],
                )
        async with ranked_lock:
            ranked_count += 1
            await deps.broadcast_fn({"type": "cv_triage_progress", "job_id": job_id,
                                     "phase": "ranking", "processed": ranked_count,
                                     "total": len(extracted)})

    await asyncio.gather(*[_rank_task(fn, tx, src) for fn, (tx, src) in extracted.items()])

    # Phase: send
    if _is_cancelled(job_id): raise asyncio.CancelledError
    storage.update_status(deps.db, job_id, JobStatus.SENDING)
    top = storage.get_top_n(deps.db, job_id, _TOP_N)
    summary = _build_summary(job.rubric, job.target_label, top,
                             storage.get_job(deps.db, job_id))
    top5_json = json.dumps([{"rank": i+1, "filename": r.filename,
                              "candidate_name": r.candidate_name, "score": r.score,
                              "match_pct": r.match_pct, "strengths": r.strengths,
                              "gaps": r.gaps, "justification": r.justification}
                             for i, r in enumerate(top)])
    storage.update_status(deps.db, job_id, JobStatus.SENDING, top5_json=top5_json)

    payload = DispatchPayload(
        chat_id=job.target_chat_id, summary=summary,
        pdfs=[DispatchPdf(rank=i+1, candidate_name=r.candidate_name or r.filename,
                          score=r.score or 0, path=staging / r.filename,
                          filename=r.filename)
              for i, r in enumerate(top)],
    )
    dispatch = await dispatch_to_whatsapp(payload, bridge_url=deps.bridge_url,
                                          sleep_between=deps.sleep_between_media)
    if dispatch.text_sent and not dispatch.failed_ranks:
        wa_status = WhatsAppStatus.ALL_SENT
    elif dispatch.text_sent:
        wa_status = WhatsAppStatus.TEXT_SENT
    else:
        wa_status = WhatsAppStatus.FAILED

    storage.update_status(deps.db, job_id, JobStatus.DONE,
                          whatsapp_status=wa_status, finished_at=_now())
    await deps.broadcast_fn({"type": "cv_triage_finished", "job_id": job_id,
                             "top5": json.loads(top5_json),
                             "whatsapp_status": wa_status.value,
                             "unreadable": storage.get_job(deps.db, job_id).unreadable})

def _build_summary(rubric: str, target_label: str, top: list, job) -> str:
    lines = [f"📄 *Triagem de CV — Top {len(top)}*", f"Vaga: {rubric[:200]}", ""]
    for i, r in enumerate(top, start=1):
        name = r.candidate_name or r.filename
        lines.append(f"*{i}. {name}* · score {r.score:.0f} · match {r.match_pct:.0f}%")
        if r.justification:
            lines.append(f"   {r.justification}")
    if job.unreadable:
        lines.append("")
        lines.append(f"⚠ {job.unreadable} CV(s) não puderam ser lidos (entrarão como anexo na lista).")
    return "\n".join(lines)
```

```python
async def resend(request: Request, job_id: int, body: ResendBody | None = None):
    """Re-runs ONLY the send phase against the stored top5. Useful when the
    bridge was offline at the original dispatch time, or to retarget another
    group/contact without re-ranking."""
    import json as _json
    db = _conn(request)
    job = storage.get_job(db, job_id)
    if not job: raise HTTPException(404)
    if job.status != JobStatus.DONE: raise HTTPException(409, "only done jobs can be resent")
    if not job.top5_json: raise HTTPException(409, "no top5 snapshot to send")
    chat_id = (body.target_chat_id if body else None) or job.target_chat_id
    top5 = _json.loads(job.top5_json)
    from .whatsapp import dispatch_to_whatsapp, DispatchPayload, DispatchPdf
    staging = Path(job.staging_dir)
    pdfs = [DispatchPdf(rank=t["rank"], candidate_name=t["candidate_name"] or t["filename"],
                        score=float(t["score"]), path=staging / t["filename"],
                        filename=t["filename"])
            for t in top5 if (staging / t["filename"]).exists()]
    summary = _build_resend_summary(job.rubric, top5)
    payload = DispatchPayload(chat_id=chat_id, summary=summary, pdfs=pdfs)
    result = await dispatch_to_whatsapp(payload, bridge_url=_bridge_url(request))
    wa = "all_sent" if result.text_sent and not result.failed_ranks \
        else ("text_sent" if result.text_sent else "failed")
    storage.update_status(db, job_id, JobStatus.DONE,
                          whatsapp_status=__import__("seraphine.cv_triage.models",
                                                     fromlist=["WhatsAppStatus"]).WhatsAppStatus(wa))
    return {"ok": result.text_sent, "media_sent": result.media_sent,
            "failed_ranks": result.failed_ranks}

def _build_resend_summary(rubric: str, top5: list[dict]) -> str:
    lines = [f"📄 *Triagem de CV — Top {len(top5)} (reenvio)*", f"Vaga: {rubric[:200]}", ""]
    for t in top5:
        name = t.get("candidate_name") or t["filename"]
        lines.append(f"*{t['rank']}. {name}* · score {t['score']:.0f}")
        if t.get("justification"): lines.append(f"   {t['justification']}")
    return "\n".join(lines)
```

```python
import shutil
import sqlite3
from datetime import datetime, timezone, timedelta
from pathlib import Path

def cleanup(conn: sqlite3.Connection, *, base_dir: Path,
            done_ttl_h: int = 24, failed_ttl_h: int = 7 * 24) -> int:
    now = datetime.now(timezone.utc)
    rows = conn.execute(
        "SELECT id, status, staging_dir, finished_at FROM cv_triage_jobs "
        "WHERE finished_at IS NOT NULL"
    ).fetchall()
    removed = 0
    for r in rows:
        finished = datetime.fromisoformat(r["finished_at"].replace("Z", "+00:00"))
        if finished.tzinfo is None:
            finished = finished.replace(tzinfo=timezone.utc)
        ttl = failed_ttl_h if r["status"] == "failed" else done_ttl_h
        if now - finished < timedelta(hours=ttl):
            continue
        staging = Path(r["staging_dir"])
        if staging.exists():
            shutil.rmtree(staging, ignore_errors=True)
            removed += 1
    return removed
```

```rust
use std::fs::File;
use std::io::{Read, Write};
use std::path::{Path, PathBuf};
use zip::write::SimpleFileOptions;
use zip::CompressionMethod;

#[tauri::command]
pub fn zip_folder_pdfs(folder_path: String) -> Result<Vec<u8>, String> {
    let folder = PathBuf::from(&folder_path);
    if !folder.is_dir() {
        return Err(format!("not a directory: {folder_path}"));
    }
    let mut buf: Vec<u8> = Vec::new();
    {
        let cursor = std::io::Cursor::new(&mut buf);
        let mut zw = zip::ZipWriter::new(cursor);
        let opts = SimpleFileOptions::default().compression_method(CompressionMethod::Deflated);
        for entry in walk_pdfs(&folder).map_err(|e| e.to_string())? {
            let rel = entry.file_name().to_string_lossy().to_string();
            zw.start_file(&rel, opts).map_err(|e| e.to_string())?;
            let mut f = File::open(entry.path()).map_err(|e| e.to_string())?;
            let mut bytes = Vec::new();
            f.read_to_end(&mut bytes).map_err(|e| e.to_string())?;
            zw.write_all(&bytes).map_err(|e| e.to_string())?;
        }
        zw.finish().map_err(|e| e.to_string())?;
    }
    Ok(buf)
}

fn walk_pdfs(dir: &Path) -> std::io::Result<Vec<std::fs::DirEntry>> {
    let mut out = Vec::new();
    for entry in std::fs::read_dir(dir)? {
        let entry = entry?;

        }
    }

}
```

```tsx
import { useState } from "react";
import { invoke } from "@tauri-apps/api/core";

import type { Chat } from "./types";

interface Props {

export function TriageWizard({ onClose, onCreated }: Props) {
  const [step, setStep] = useState<1 | 2 | 3>(1);
  const [folder, setFolder] = useState<string | null>(null);
  const [pdfCount, setPdfCount] = useState(0);
  const [rubric, setRubric] = useState("");
  const [chats, setChats] = useState<Chat[]>([]);
  const [chatId, setChatId] = useState<string | null>(null);
  const [search, setSearch] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  async function pickFolder() {
    const sel = await open({ directory: true, multiple: false });
    if (typeof sel !== "string") return;
    setFolder(sel);
    // ask Rust for the PDF count without zipping
    const names = (await invoke<string[]>("list_dir_local", { path: sel })) || [];
    setPdfCount(names.filter(n => n.toLowerCase().endsWith(".pdf") && !n.startsWith("._") && n !== ".DS_Store").length);
  }

  async function loadChats() {
    setChats(await listChats());
  }

  async function submit() {
    if (!folder || !chatId || rubric.trim().length < 30) return;
    setSubmitting(true);
    setErr(null);
    try {
      const zipBytes = await invoke<number[]>("zip_folder_pdfs", { folderPath: folder });
      const label = chats.find(c => c.id === chatId)?.name ?? chatId;
      const { job_id } = await createJob({
        zip: new Uint8Array(zipBytes),
        rubric: rubric.trim(),
        target_chat_id: chatId,
        target_label: label,
      });
      onCreated(job_id);
      onClose();
    } catch (e: any) {
      setErr(String(e?.message ?? e));
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <div className="modal">
      <div className="modal-card cv-triage-wizard">
        <header>
          <h2>Triagem de CV — passo {step}/3</h2>
          <button onClick={onClose}>×</button>
        </header>
        {step === 1 && (
          <section>
            <button onClick={pickFolder}>Escolher pasta…</button>
            {folder && <p>{folder} · {pdfCount} PDFs detectados</p>}
            <footer>
              <button disabled={!folder || pdfCount === 0} onClick={() => setStep(2)}>Próximo</button>
            </footer>
          </section>
        )}
        {step === 2 && (
          <section>
            <textarea
              placeholder="Cargo, nível, skills obrigatórias, diferenciais, localização..."
              value={rubric}
              onChange={e => setRubric(e.target.value)}
              rows={8}
            />
            <p>{rubric.trim().length}/30 mínimo</p>
            <footer>
              <button onClick={() => setStep(1)}>Voltar</button>
              <button disabled={rubric.trim().length < 30}
                onClick={async () => { await loadChats(); setStep(3); }}>Próximo</button>
            </footer>
          </section>
        )}
        {step === 3 && (
          <section>
            <input placeholder="Buscar grupo/contato..."
                   value={search} onChange={e => setSearch(e.target.value)} />
            <ul className="chat-list">
              {chats
                .filter(c => c.name.toLowerCase().includes(search.toLowerCase()))
                .sort((a, b) => Number(b.isGroup) - Number(a.isGroup))
                .map(c => (

              </button>
            </footer>
          </section>
        )}
      </div>
    </div>
  );
}
```

```tsx
import { useEffect, useState } from "react";
import { useNavigate } from "react-router-dom"; // or your router
import { getJob } from "./api";
import type { JobDetail, ProgressFrame, FinishedFrame } from "./types";
import { useWebSocket } from "../ws/useWebSocket"; // adapt to your WS hook

interface Props { jobId: number; }

export function JobProgressCard({ jobId }: Props) {
  const [job, setJob] = useState<JobDetail | null>(null);
  const [collapsed, setCollapsed] = useState(false);
  const ws = useWebSocket();
  const navigate = useNavigate();

  useEffect(() => { getJob(jobId).then(setJob).catch(() => {}); }, [jobId]);

  useEffect(() => {
    if (!ws) return;
    const handler = (msg: any) => {
      if ((msg.type === "cv_triage_progress" || msg.type === "cv_triage_finished")
          && msg.job_id === jobId) {
        // optimistic merge — refresh full state for finished frame
        if (msg.type === "cv_triage_finished") getJob(jobId).then(setJob);
        else setJob(prev => prev ? { ...prev, processed_cvs: msg.processed,
                                     total_cvs: msg.total,
                                     ocr_used_count: msg.ocr_used_count,
                                     vision_used_count: msg.vision_used_count,
                                     status: msg.phase as any } : prev);
      }
    };
    ws.addMessageListener(handler);
    return () => ws.removeMessageListener(handler);
  }, [ws, jobId]);

  if (!job) return <div className="cv-card loading">Carregando…</div>;
  const pct = job.total_cvs ? Math.round(100 * job.processed_cvs / job.total_cvs) : 0;
  const label = phaseLabel(job.status, job.ocr_used_count, job.vision_used_count);

  if (collapsed) {
    return (
      <div className="cv-card collapsed" onClick={() => setCollapsed(false)}>
        📄 Triagem · {job.processed_cvs}/{job.total_cvs} · {job.status}
      </div>
    );
  }

  return (
    <div className="cv-card">
      <header onClick={() => setCollapsed(true)}>
        <span>📄 Triagem de CV — {job.total_cvs} currículos</span>
        <button onClick={(e) => { e.stopPropagation(); navigate(`/jobs/${job.id}`); }}>▸</button>
      </header>
      {job.status !== "done" && job.status !== "failed" && job.status !== "cancelled" && (
        <>
          <div className="bar"><div className="fill" style={{ width: `${pct}%` }} /></div>
          <p>{label}</p>
        </>
      )}
      {job.status === "done" && job.top5 && (
        <ol className="top5">
          {job.top5.map(t => (
            <li key={t.rank}>
              <strong>{t.rank}. {t.candidate_name ?? t.filename}</strong> · score {t.score.toFixed(0)}
            </li>
          ))}
        </ol>
      )}
      {job.status === "failed" && (
        <p className="error">⚠ {job.error}</p>
      )}
      <p className="target">→ {job.target_label}</p>
    </div>
  );
}
```

```tsx
}

function groupByDay(jobs: JobSummary[]): { label: string; items: JobSummary[] }[] {
  const out: Record<string, JobSummary[]> = {};
  const today = new Date(); today.setHours(0,0,0,0);
  const yesterday = new Date(today.getTime() - 86400_000);
  const weekAgo = new Date(today.getTime() - 7 * 86400_000);
  for (const j of jobs) {
    const d = new Date(j.created_at);
    const label = d >= today ? "Hoje"
      : d >= yesterday ? "Ontem"
      : d >= weekAgo ? "Esta semana" : "Mais antigo";
    (out[label] ||= []).push(j);
  }
  return ["Hoje","Ontem","Esta semana","Mais antigo"]
    .filter(l => out[l]?.length)
    .map(label => ({ label, items: out[label] }));
}
```

## Out of Scope (do NOT implement)

These are explicitly excluded from the spec and must not be added during implementation:

- Cross-job candidate deduplication / talent pool persistence.
- Editing rubric and re-ranking in place.
- `.docx`/`.txt`/`.rtf` parsing — only `.pdf`.
- Multi-language OCR — only PT-BR.
- Anthropic Message Batches API.
- A generic "remote folder read" reverse-channel (the rejected Option C from brainstorming).

## Implementation order summary

The phases are ordered to keep every phase shippable on its own. After Phase 0 the bridge is independently usable for media sends. After Phase 6 the backend is fully callable via HTTP (frontend not strictly required). The frontend phases (7–8) layer UI on top. Phase 9 is verification only.

If splitting work across multiple sessions, natural pause points are:
- After **Task 4** (schema lives in DB, nothing else risked).
- After **Task 9** (backend libs done, no routes yet — can be exercised by a quick scratch script).
- After **Task 14** (backend complete, can be driven from `curl` for the first proof-of-life).
- After **Task 18** (chat card live, can already demo end to end with the "Trabalhos" tab postponed).
